        // ========================== SOUND SYSTEM ==========================
        function playSound(type) {
            const ctx = new (window.AudioContext || window.webkitAudioContext)();
            if (type === "dot") {
                const osc = ctx.createOscillator();
                osc.frequency.value = 820 + Math.random() * 620;
                const gain = ctx.createGain();
                gain.gain.value = 0.28;
                const filter = ctx.createBiquadFilter();
                filter.type = 'lowpass';
                filter.frequency.value = 1800;
                osc.connect(filter).connect(gain).connect(ctx.destination);
                osc.start();
                osc.stop(ctx.currentTime + 0.09);
            } 
            else if (type === "correct") {
                let t = ctx.currentTime;
                [880, 1120, 1380, 1680].forEach((f, i) => {
                    const o = ctx.createOscillator();
                    o.frequency.setValueAtTime(f, t);
                    const g = ctx.createGain();
                    g.gain.setValueAtTime(0.42, t);
                    g.gain.exponentialRampToValueAtTime(0.001, t + 0.85);
                    o.connect(g).connect(ctx.destination);
                    o.start(t);
                    o.stop(t + 0.9);
                    t += 0.11;
                });
            } 
            else if (type === "wrong") {
                const osc = ctx.createOscillator();
                osc.frequency.setValueAtTime(320, ctx.currentTime);
                const gain = ctx.createGain();
                gain.gain.setValueAtTime(0.35, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.65);
                osc.connect(gain).connect(ctx.destination);
                osc.start();
                osc.stop(ctx.currentTime + 0.65);
            } 
            else if (type === "win") {
                let t = ctx.currentTime;
                [680, 880, 1100, 1350, 1680].forEach(f => {
                    const o = ctx.createOscillator();
                    o.frequency.setValueAtTime(f, t);
                    const g = ctx.createGain();
                    g.gain.setValueAtTime(0.38, t);
                    g.gain.exponentialRampToValueAtTime(0.001, t + 1.35);
                    o.connect(g).connect(ctx.destination);
                    o.start(t);
                    o.stop(t + 1.4);
                    t += 0.13;
                });
            }
        }

        // ========================== CHAPTER LOADING ==========================
        function loadChapterImage(chapter) {
            if (currentPlane) scene.remove(currentPlane);
            pieces.forEach(p => scene.remove(p));
            pieces = [];

            const loader = new THREE.TextureLoader();
            loader.load(chapterImages[chapter-1], (texture) => {
                const aspect = texture.image.width / texture.image.height;
                const geo = new THREE.PlaneGeometry(7.8, 7.8 / aspect);

                for (let i = 0; i < 16; i++) {
                    const piece = new THREE.Mesh(geo, new THREE.MeshPhongMaterial({
                        map: texture,
                        transparent: true,
                        opacity: 0.88,
                        side: THREE.DoubleSide,
                        depthTest: true
                    }));
                    piece.scale.set(0.31, 0.31, 1);
                    piece.position.set((Math.random()-0.5)*9.5, (Math.random()-0.5)*7.5, 0.2 + Math.random()*0.3);
                    piece.rotation.z = (Math.random() - 0.5) * 1.6;
                    piece.userData = { baseX: piece.position.x, baseY: piece.position.y };
                    scene.add(piece);
                    pieces.push(piece);
                }
            }, undefined, () => console.warn("Image load failed for chapter", chapter));
        }

// ========================== PARTICLES ==========================
        function createFlyingDots(count) {
            const safeCount = Math.min(count, 120); // prevent lag
            for (let i = 0; i < safeCount; i++) {
                const p = new THREE.Mesh(
                    new THREE.SphereGeometry(0.038, 8, 8),
                    new THREE.MeshBasicMaterial({ color: 0xffeebb })
                );
                p.position.set(
                    (Math.random()-0.5)*13,
                    (Math.random()-0.5)*9,
                    (Math.random()-0.5)*5
                );
                p.velocity = new THREE.Vector3(
                    (Math.random()-0.5)*1.4,
                    (Math.random()-0.5)*1.4,
                    (Math.random()-0.5)*0.7
                );
                scene.add(p);
                particles.push(p);
            }
        }

        // ========================== DOT BUTTON ==========================
        document.getElementById('dotBtn').addEventListener('click', () => {
            if (isProcessing) return;
            playSound("dot");
            createFlyingDots(38);
            
            dotsPlaced += 10000;
            const target = chapterTargets[currentChapter-1];
            const progress = Math.min(dotsPlaced / target, 1);

            // Shatter animation
            pieces.forEach(piece => {
                piece.position.x *= 0.84;
                piece.position.y *= 0.84;
                piece.rotation.z *= 0.79;
            });

            document.getElementById('progressBar').style.width = (progress * 100) + '%';

            if (dotsPlaced >= target) {
                isProcessing = true;
                completeChapter();
            }
        });

        function completeChapter() {
            totalScore += 100;
            document.getElementById('scoreValue').textContent = totalScore;
            createFlyingDots(95);

            // Assemble pieces
            pieces.forEach((p, i) => {
                p.position.set(0, 0, 0.1 + i * 0.01);
                p.rotation.z = 0;
                p.scale.setScalar(1);
            });

            setTimeout(() => {
                playSound("correct");
                attempts = 2;
                
                document.getElementById('modalChapter').textContent = String(currentChapter).padStart(2, '0');
                document.getElementById('hint').textContent = Hint: Around ${Math.round(chapterTargets[currentChapter-1] / 1000)}K dots;
                document.getElementById('attemptsLeft').textContent = "2 attempts remaining";
                document.getElementById('resultMessage').innerHTML = '';
                document.getElementById('guessInput').value = '';
                document.getElementById('guessModal').style.display = 'flex';
                document.getElementById('guessInput').focus();
                
                isProcessing = false;
            }, 1650);
        }

        // ========================== GUESS HANDLING ==========================
        window.submitGuess = function() {
            const guessInput = document.getElementById('guessInput');
            const guess = parseInt(guessInput.value) || 0;
            const actual = chapterTargets[currentChapter-1];
            const resultMsg = document.getElementById('resultMessage');
            
            attempts--;

            if (guess === actual) {
                passedChapters++;
                totalScore += 10;
                document.getElementById('scoreValue').textContent = totalScore;
                playSound("correct");
                resultMsg.innerHTML = "CORRECT";
                resultMsg.className = "correct";
                document.getElementById('submitBtn').style.display = "none";
                setTimeout(nextChapter, 1650);
            } 
            else if (attempts > 0) {
                playSound("wrong");
                resultMsg.innerHTML = "WRONG";
                resultMsg.className = "wrong";
                document.getElementById('attemptsLeft').
textContent = ${attempts} attempts remaining;
                
                setTimeout(() => {
                    resultMsg.innerHTML = '';
                    guessInput.value = '';
                    guessInput.focus();
                }, 1350);
            } 
            else {
                playSound("wrong");
                resultMsg.innerHTML = WRONG<br><small>The number was ${actual.toLocaleString()}</small>;
                resultMsg.className = "wrong";
                document.getElementById('submitBtn').style.display = "none";
                setTimeout(nextChapter, 2450);
            }
        };

        function nextChapter() {

        // ========================== ANIMATION LOOP ==========================
        function animate() {
            requestAnimationFrame(animate);

            const center = new THREE.Vector3(0, 0, 0);
            
            for (let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i];
                const dir = center.clone().sub(p.position).normalize();
                p.velocity.addScaledVector(dir, 0.032);
                p.position.add(p.velocity);
                p.velocity.multiplyScalar(0.935);

                if (p.position.length() < 0.55) {
                    scene.remove(p);
                    particles.splice(i, 1);
                }
            }

            renderer.render(scene, camera);
        }
